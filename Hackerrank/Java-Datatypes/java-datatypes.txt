public static void main(String []argh) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();

        for (int i = 0; i < t; i++) {
            try {
                long x = sc.nextLong();
                System.out.println(x + " can be fitted in:");
                
                // 8-bit signed integer
                if (x >= -128 && x <= 127) {
                    System.out.println("* byte");
                }
                // 16-bit signed integer
                if (x >= Byte.MIN_VALUE && x <= Byte.MAX_VALUE) { // already handled above, 
                but keeping hierarchy clear
                    // The problem already prints byte if it fits. 
                }
                if (x >= Short.MIN_VALUE && x <= Short.MAX_VALUE) {
                    System.out.println("* short");
                }
                // 32-bit signed integer
                if (x >= Integer.MIN_VALUE && x <= Integer.MAX_VALUE) {
                    System.out.println("* int");
                }
                // 64-bit signed integer (If it successfully parsed as a long, it fits in 
                long)
                if (x >= Long.MIN_VALUE && x <= Long.MAX_VALUE) {
                    System.out.println("* long");
                }
                
            } catch (Exception e) {
                // If the number is too massive for a long, nextLong() throws an exception
                System.out.println(sc.next() + " can't be fitted anywhere.");
            }
        }
    }
}
